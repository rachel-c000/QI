/* ===================================================== QI SHARE MODAL — logic Assumes the following are already defined on the page: - keys, colors, bgMap, scores, dominant (or computed below) - animal object: { name, img, desc } - top3 array (top 3 emotion keys) - tracks array for music page (optional) ===================================================== */ /* ── colour maps (safe to redeclare if not on page) ── */ var _qi_colors = typeof colors !== 'undefined' ? colors : { happy:'#FFD71E', sad:'#2264DC', angry:'#E8251A', peace:'#50D2A0', love:'#FF64A0', confused:'#A050DC', fear:'#3d1a6e', disgust:'#6E8C32' }; var _qi_bgMap = typeof bgMap !== 'undefined' ? bgMap : { happy:'#f5f0d8', sad:'#d8e4f5', angry:'#f5dbd8', peace:'#c8f0e4', love:'#f5d8e8', confused:'#ede0f5', fear:'#ddd8f5', disgust:'#dff0d8' }; var _qi_keys = ['happy','sad','angry','peace','love','confused','fear','disgust']; /* ── read scores from sessionStorage ── */ var _qi_scores = {}; _qi_keys.forEach(function(k) { _qi_scores[k] = parseInt(sessionStorage.getItem(k)) || 0; }); var _qi_dominant = _qi_keys.reduce(function(a,b) { return _qi_scores[a] >= _qi_scores[b] ? a : b; }); var _qi_top3 = _qi_keys.slice().sort(function(a,b) { return _qi_scores[b] - _qi_scores[a]; }).slice(0,3); /* ── user id ── */ var _qi_userId = parseInt(sessionStorage.getItem('qi_user_id')) || 1; /* ── animal data from sessionStorage (set this when animal is determined) ── On each results page where animal is computed, add: sessionStorage.setItem('qi_animal_name', animal.name); sessionStorage.setItem('qi_animal_desc', animal.desc); sessionStorage.setItem('qi_animal_img', animal.img); ── */ var _qi_animal = { name: sessionStorage.getItem('qi_animal_name') || '', desc: sessionStorage.getItem('qi_animal_desc') || '', img: sessionStorage.getItem('qi_animal_img') || '' }; /* ── playlist from sessionStorage (set on music page) ── On music results page, after generating playlist, add: sessionStorage.setItem('qi_tracks', JSON.stringify(trackArray)); where trackArray = [{title, artist}, ...] ── */ var _qi_tracks = []; try { _qi_tracks = JSON.parse(sessionStorage.getItem('qi_tracks') || '[]'); } catch(e) {} /* ── user's own reflection ── On reflect page, after submitPost(), also run: sessionStorage.setItem('qi_reflection', text); ── */ var _qi_reflection = sessionStorage.getItem('qi_reflection') || ''; /* ───────────────────────────────────────── POPULATE CARD ───────────────────────────────────────── */ function populateCard() { var card = document.getElementById('qi-summary-card'); if (!card) return; /* background */ card.style.background = _qi_bgMap[_qi_dominant] || '#f5f0d8'; /* user id dot + label */ var dot = document.getElementById('card-dot'); if (dot) dot.style.background = _qi_colors[_qi_dominant]; var idEl = document.getElementById('card-id'); if (idEl) idEl.textContent = 'No.' + _qi_userId; /* dominant */ var domEl = document.getElementById('card-dominant-value'); if (domEl) domEl.textContent = _qi_dominant; /* top 3 tiles as chips */ var tilesEl = document.getElementById('card-tiles'); if (tilesEl) { tilesEl.innerHTML = ''; _qi_top3.forEach(function(e) { var chip = document.createElement('span'); chip.className = 'card-tile-chip'; chip.textContent = e; chip.style.borderColor = _qi_colors[e]; tilesEl.appendChild(chip); }); } /* animal */ var aImg = document.getElementById('card-animal-img'); var aName = document.getElementById('card-animal-name'); var aDesc = document.getElementById('card-animal-desc'); if (aImg) aImg.src = _qi_animal.img; if (aName) aName.textContent = _qi_animal.name; if (aDesc) aDesc.textContent = _qi_animal.desc; /* playlist — show up to 5 tracks */ var tracksEl = document.getElementById('card-tracks'); if (tracksEl) { tracksEl.innerHTML = ''; var showTracks = _qi_tracks.slice(0, 5); if (showTracks.length) { showTracks.forEach(function(t, i) { var div = document.createElement('div'); div.className = 'card-track'; div.textContent = (i+1) + '. ' + (t.title || t.name || t) + (t.artist ? ' — ' + t.artist : ''); tracksEl.appendChild(div); }); } else { var pl = document.getElementById('card-playlist-section'); if (pl) pl.style.display = 'none'; } } /* reflection */ if (_qi_reflection) { var refSection = document.getElementById('card-reflection-section'); var refText = document.getElementById('card-reflection-text'); if (refSection) refSection.style.display = ''; if (refText) refText.textContent = '"' + _qi_reflection + '"'; } /* date */ var dateEl = document.getElementById('card-date'); if (dateEl) { var d = new Date(); dateEl.textContent = d.toLocaleDateString('en-AU', { day:'numeric', month:'short', year:'numeric' }); } } /* ───────────────────────────────────────── BUILD SHARE URL Encodes session state as URL params so the phone landing page can reconstruct the card ───────────────────────────────────────── */ function buildShareUrl() { var base = window.location.origin + window.location.pathname .replace(/\/[^/]*$/, '/share-card.html'); /* → /share-card.html */ var params = new URLSearchParams(); params.set('uid', _qi_userId); params.set('dominant', _qi_dominant); params.set('top3', _qi_top3.join(',')); params.set('animal', _qi_animal.name); params.set('adesc', _qi_animal.desc.slice(0, 120)); /* trim for URL */ params.set('aimg', _qi_animal.img); if (_qi_tracks.length) { var trackStr = _qi_tracks.slice(0,5).map(function(t) { return (t.title || t.name || t) + (t.artist ? '|' + t.artist : ''); }).join(';;'); params.set('tracks', trackStr); } if (_qi_reflection) { params.set('ref', _qi_reflection.slice(0, 200)); } _qi_keys.forEach(function(k) { params.set(k, _qi_scores[k]); }); return base + '?' + params.toString(); } /* ───────────────────────────────────────── GENERATE QR CODE ───────────────────────────────────────── */ var _qi_qr_generated = false; function generateQR(url) { if (_qi_qr_generated) return; var canvas = document.getElementById('qi-qr-canvas'); if (!canvas || typeof QRCode === 'undefined') return; /* QRCode.js into an existing canvas */ try { var qr = new QRCode(canvas, { text: url, width: 140, height: 140, colorDark: '#3d3228', colorLight: '#faf7f0', correctLevel: QRCode.CorrectLevel.M }); _qi_qr_generated = true; } catch(e) { /* fallback: show URL only */ console.warn('QR generation failed:', e); } var urlEl = document.getElementById('qi-share-url'); if (urlEl) urlEl.textContent = url.length > 80 ? url.slice(0,77)+'…' : url; } /* ───────────────────────────────────────── MODAL OPEN / CLOSE ───────────────────────────────────────── */ function openQiModal() { populateCard(); var url = buildShareUrl(); generateQR(url); var backdrop = document.getElementById('qi-modal-backdrop'); if (backdrop) backdrop.classList.add('open'); document.body.style.overflow = 'hidden'; } function closeQiModal() { var backdrop = document.getElementById('qi-modal-backdrop'); if (backdrop) backdrop.classList.remove('open'); document.body.style.overflow = ''; } function handleBackdropClick(e) { if (e.target === document.getElementById('qi-modal-backdrop')) closeQiModal(); } /* close on Escape */ document.addEventListener('keydown', function(e) { if (e.key === 'Escape') closeQiModal(); }); /* ───────────────────────────────────────── DOWNLOAD CARD AS IMAGE ───────────────────────────────────────── */ function downloadCard() { var card = document.getElementById('qi-summary-card'); if (!card || typeof html2canvas === 'undefined') { alert('Download not available. Try scanning the QR code instead.'); return; } html2canvas(card, { scale: 2, /* retina quality */ useCORS: true, /* allow cross-origin images (animal imgs) */ backgroundColor: null, logging: false }).then(function(canvas) { var link = document.createElement('a'); link.download = 'qi-card-no' + _qi_userId + '.png'; link.href = canvas.toDataURL('image/png'); link.click(); }).catch(function(err) { console.error('html2canvas error:', err); alert('Could not generate image. Try scanning the QR code instead.'); }); } /* ───────────────────────────────────────── HELPER — call this from each results page after the animal is determined, to persist it: qiPersistAnimal(animal); // animal = { name, img, desc } qiPersistTracks(tracks); // tracks = [{title, artist}, ...] qiPersistReflection(text); // text = string ───────────────────────────────────────── */ function qiPersistAnimal(a) { if (!a) return; sessionStorage.setItem('qi_animal_name', a.name || ''); sessionStorage.setItem('qi_animal_desc', a.desc || ''); sessionStorage.setItem('qi_animal_img', a.img || ''); _qi_animal = { name: a.name, desc: a.desc, img: a.img }; } function qiPersistTracks(arr) { if (!Array.isArray(arr)) return; sessionStorage.setItem('qi_tracks', JSON.stringify(arr)); _qi_tracks = arr; } function qiPersistReflection(text) { if (!text) return; sessionStorage.setItem('qi_reflection', text); _qi_reflection = text; }
